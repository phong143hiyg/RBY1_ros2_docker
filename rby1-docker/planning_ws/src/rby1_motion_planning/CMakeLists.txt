cmake_minimum_required(VERSION 3.16)
project(rby1_motion_planning)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp std_msgs std_srvs sensor_msgs geometry_msgs visualization_msgs moveit_msgs moveit_core moveit_ros_planning moveit_ros_planning_interface tf2_eigen)
  find_package(${dep} REQUIRED)
endforeach()
find_package(pluginlib REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(moveit_task_constructor_core QUIET)
set(deps rclcpp std_msgs std_srvs sensor_msgs geometry_msgs visualization_msgs moveit_msgs moveit_core moveit_ros_planning moveit_ros_planning_interface tf2_eigen)
add_library(planning_only_controller_manager SHARED src/planning_only_controller_manager.cpp)
ament_target_dependencies(planning_only_controller_manager moveit_core pluginlib rclcpp)
pluginlib_export_plugin_description_file(moveit_core config/planning_controller_plugin.xml)
add_library(planning_common src/scene.cpp src/validator.cpp)
target_include_directories(planning_common PUBLIC include)
ament_target_dependencies(planning_common ${deps})
target_link_libraries(planning_common yaml-cpp nlohmann_json::nlohmann_json)
add_executable(planning_worker src/planning_worker.cpp)
target_link_libraries(planning_worker planning_common)
ament_target_dependencies(planning_worker ${deps})
if(moveit_task_constructor_core_FOUND)
  target_sources(planning_worker PRIVATE src/mtc_task.cpp)
  target_compile_definitions(planning_worker PRIVATE RBY1_HAS_MTC=1)
  ament_target_dependencies(planning_worker moveit_task_constructor_core)
endif()
add_executable(scene_loader src/scene_loader.cpp)
target_link_libraries(scene_loader planning_common)
ament_target_dependencies(scene_loader ${deps})
install(TARGETS planning_only_controller_manager planning_common planning_worker scene_loader ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config fixtures DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/planning_service.py scripts/planning_contract_service.py scripts/planning_cli.py scripts/run_scenarios.py scripts/capture_environment.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/protocol_core.py scripts/contract_v1.py DESTINATION lib/${PROJECT_NAME})
option(RBY1_ENABLE_INTEGRATION_TESTS "Start isolated fake launch during colcon test" OFF)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_pytest_test(protocol_core test/test_protocol.py APPEND_ENV PYTEST_DISABLE_PLUGIN_AUTOLOAD=1)
  ament_add_pytest_test(contract_v1 test/test_contract_v1.py APPEND_ENV PYTEST_DISABLE_PLUGIN_AUTOLOAD=1)
  ament_add_gtest(validator_test test/validator_test.cpp)
  target_link_libraries(validator_test planning_common)
  ament_target_dependencies(validator_test ${deps})
  if(RBY1_ENABLE_INTEGRATION_TESTS)
    find_package(launch_testing_ament_cmake REQUIRED)
    add_launch_test(test/test_planning_launch.py TIMEOUT 330)
  endif()
endif()
ament_package()
